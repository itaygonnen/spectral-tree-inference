\section{Proof of the Laplacian sampling error}\label{app:comp2}

We bound $\opnorm{\EL}\le\opnorm{\ED}+\opnorm{\ES}$, where $\ES=\hat S-S$ and
$\ED=\hat D-D$, and show $\opnorm{\ES}$ dominates. We first record three structural
facts about $S$.

\begin{proposition}[Rank-2 dominance of the rows]\label{prop:B-rank2}
Under \Cref{asm:cbm}, $S$ is rank $2$ and its row mass is captured by its leading
rank-$2$ decomposition $U\Lambda U^\top$:
$\max_i\norm{S_{i,\cdot}}_2\le\max_i\norm{e_i^\top U\Lambda U^\top}_2$, and
$\max_i\norm{e_i^\top U\Lambda U^\top}_2^2\le\lambda_{\max}^2\max_i\norm{e_i^\top U}_2^2$.
\end{proposition}
\begin{proof}
The first inequality is immediate since $S=U\Lambda U^\top$ under the CBM. For the
second, put $x^\top=e_i^\top U$; orthonormality of $U$'s columns preserves norm
under $U^\top$, so with $\Lambda=\operatorname{diag}(\lambda_1,\lambda_2)$,
$\norm{x^\top\Lambda U^\top}_2^2=\lambda_1^2x_1^2+\lambda_2^2x_2^2\le
\lambda_{\max}^2\norm{e_i^\top U}_2^2$.
\end{proof}

\begin{proposition}[Spectral scaling]\label{prop:B-scale}
Under \Cref{asm:cbm,asm:bounded}, $\lambda_{\max}(S)\le m\,\Smax$.
\end{proposition}
\begin{proof}
$S$ is non-negative, so by Perron--Frobenius its spectral radius is at most the
maximum row sum, $\lambda_{\max}(S)\le\max_i\sum_j S_{ij}\le m\Smax$.
\end{proof}

\begin{proposition}[Row-norm / coherence bound]\label{prop:B-rownorm}
Under \Cref{asm:cbm}, $\max_i\norm{U_{i,:}}_2^2=\tfrac{2\coh}{m}=\tfrac{1+\eta}{m}$.
\end{proposition}
\begin{proof}
Immediate from \eqref{eq:coherence_definition} and $\coh=(1+\eta)/2$ (\Cref{prop:coherence}).
\end{proof}

\begin{lemma}[Laplacian spectral error via matrix Bernstein]\label{lem:B-bernstein}
Under \Cref{asm:cbm,asm:bounded,asm:sampling}, there are constants $C,C_0$ such
that if $p\ge C_0\,\tfrac{(1+\eta)\log m}{m}$ then \hp
\[
   \opnorm{\EL}\le C\sqrt{\frac{m(1+\eta)\log m}{p}}.
\]
\end{lemma}
\begin{proof}
\emph{Bounding $\opnorm{\ES}$.} Write $\ES=\sum_{i\le j}Z^{(ij)}$, where $Z^{(ij)}$
is supported on the symmetric pair $(i,j)$ with value $(\tfrac1p\Omega_{ij}-1)S_{ij}$;
the summands are independent and mean zero. The scale is
$\opnorm{Z^{(ij)}}\le\max(\tfrac1p-1,1)S_{ij}\le\Smax/p=:R$. For the variance proxy,
$\E[(Z^{(ij)}_{ij})^2]=\tfrac{p(1-p)}{p^2}S_{ij}^2\le\tfrac1pS_{ij}^2$, so
$\sigma^2\le\tfrac1p\max_i\norm{S_{i,\cdot}}_2^2$. Combining
\Cref{prop:B-rank2,prop:B-scale,prop:B-rownorm} with
$\max_i\norm{e_i^\top U}_2^2=2\coh/m$,
\[
   \max_i\norm{S_{i,\cdot}}_2^2\le\lambda_{\max}^2\tfrac{2\coh}{m}
   \le(m\Smax)^2\tfrac{2\coh}{m}=2m\Smax^2\coh,
   \qquad \sigma^2\le\tfrac2p m\Smax^2\coh.
\]
Matrix Bernstein,
$\Prob(\opnorm{\ES}\ge t)\le2m\exp(\tfrac{-t^2/2}{\sigma^2+Rt/3})$, gives in the
variance-dominated regime, with probability $1-\epsilon$,
$\opnorm{\ES}\le2\Smax\sqrt{m\,\coh\,\ln(2m/\epsilon)/p}$.

\emph{Bounding $\opnorm{\ED}$.} $\ED$ is diagonal with
$(\ED)_{ii}=\sum_j(\tfrac1p\Omega_{ij}-1)S_{ij}$; a scalar Bernstein with
$\sigma_D^2\le\tfrac1p mS_{\max}^2$ and $R_D=\Smax/p$ gives
$\opnorm{\ED}\lesssim\Smax\sqrt{m\ln(2m/\epsilon)/p}$.

\emph{Synthesis.} Since $\opnorm{\ES}/\opnorm{\ED}=\Theta(\sqrt{\coh})\ge1$, $\ES$
dominates. Substituting $\coh=(1+\eta)/2$ (\Cref{prop:coherence}),
\[
   \opnorm{\EL}\le\opnorm{\ES}+\opnorm{\ED}\le\tilde C\sqrt{\frac{m(1+\eta)\log m}{p}},
\]
with $\tilde C$ absorbing $\Smax$ and absolute constants.
\end{proof}
