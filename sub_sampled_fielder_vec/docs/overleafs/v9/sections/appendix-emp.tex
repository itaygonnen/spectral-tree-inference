\subsection*{Rounding-rule sensitivity}
\Cref{sec:empirical} restricts the main text to $k$-means, since a single rounding
rule cannot support a claim about rounding rules in general. \Cref{fig:operator_sensitivity}
reruns the same $\eta$-pooled sweep of \Cref{fig:pstar_gen} with two further
rules applied to the same sampled Fiedler vector: the sign rule that
\Cref{thm:main-sim} analyzes, and a $\sigma_2$-gap rule. The sign and $k$-means
operators decay as $\approx n^{-1}$, parallel to the $\log n/n$ reference, while
the $\sigma_2$-gap rule saturates --- so the $\log n/n$ scaling is a property of
the spectral signal, not of the particular rounding rule. This is methodology
supporting the choice of $k$-means as the main-text operator, not additional
evidence for \Cref{thm:main-sim}, which concerns the sign rule alone.

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\textwidth]{pstar_gen_3operators.png}
    \caption{\textbf{Generated (Kingman), rounding-rule sensitivity.}
    $\hat p^\star$ (measured NMI $\ge0.95$) vs $n$ on the $\eta$-pooled samples
    of \Cref{fig:pstar_gen}, one panel per $\eta\in\{1,5,10,15\}$, for three
    rounding rules. Dashed = free power-law fit $\hat p^\star\propto n^{b}$;
    dotted = the $C\log n/n$ reference. The Fiedler-sign (grey) and $k$-means
    (red) operators follow $b\approx-1$; the $\sigma_2$-gap operator (blue)
    saturates.}
    \label{fig:operator_sensitivity}
\end{figure}
